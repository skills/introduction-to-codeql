cmake_minimum_required(VERSION 3.30)
project(swe_bookstore VERSION 1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "BOOKSTORE|HOLD|native Windows MSVC required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
add_library(bookstore_core STATIC src/bookstore.cpp)
target_include_directories(bookstore_core PUBLIC include)
target_compile_options(bookstore_core PRIVATE /O2 /W4 /WX /permissive- /EHsc /utf-8)
target_compile_definitions(bookstore_core PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(bookstore src/main.cpp)
target_link_libraries(bookstore PRIVATE bookstore_core ws2_32)
target_compile_options(bookstore PRIVATE /O2 /W4 /WX /permissive- /EHsc /utf-8)
target_compile_definitions(bookstore PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
include(CTest)
if(BUILD_TESTING)
  add_executable(bookstore_tests tests/bookstore_tests.cpp)
  target_link_libraries(bookstore_tests PRIVATE bookstore_core)
  target_compile_options(bookstore_tests PRIVATE /O2 /W4 /WX /permissive- /EHsc /utf-8)
  add_test(NAME bookstore_core COMMAND bookstore_tests)
endif()
